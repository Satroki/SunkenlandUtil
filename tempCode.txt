	private void UpdateSwimming()
	{
		depth = GetViewerHeightAboveWater();
		if (depth < -1.9f)
		{
			FPSRigidBodyWalker.code.characterController.height = 0.5f;
			FPSRigidBodyWalker.code.characterController.center = new Vector3(0f, 0.7f, 0f);
			FPSRigidBodyWalker.code.characterController.stepOffset = 0.01f;
		}
		else
		{
			FPSRigidBodyWalker.code.ChangeCharacterControllerState();
			FPSRigidBodyWalker.code.characterController.stepOffset = 0.7f;
			FPSRigidBodyWalker.code.characterController.center = new Vector3(0f, 0.2f, 0f);
		}
		if (depth < inWaterDepth || (bool)CurAquarium)
		{
			if (!FPSRigidBodyWalker.code.isInWater)
			{
				FPSRigidBodyWalker.code.isInWater = true;
				FPSRigidBodyWalker.code.ResetParent();
				if (!FPSPlayer.code.FPSWalkerComponent.isGrounded && drivingObject == null && seatObject == null)
				{
					RM.code.PlayOneShot(sndEnterWater, UnityEngine.Random.Range(0.9f, 1.1f));
				}
			}
		}
		else
		{
			FPSRigidBodyWalker.code.isInWater = false;
			FPSRigidBodyWalker.code.isFloating = false;
		}
		if (MathF.Abs(depth) - MathF.Abs(ArrivedDepth) >= 10f)
		{
			ArrivedDepth = depth;
			Global.code.EventChannelManager.DiveDepthEventChannel.Raise(depth);
		}
		if (depth < -60f && depth > -80f)
		{
			if (depthAirConsumptionRatio != 1.2f)
			{
				depthAirConsumptionRatio = 1.2f;
			}
		}
		else if (depth < -80f)
		{
			if (depthAirConsumptionRatio != 1.4f)
			{
				depthAirConsumptionRatio = 1.4f;
			}
		}
		else
		{
			depthAirConsumptionRatio = 1f;
		}
		if (depth < -0.1f || (bool)CurAquarium)
		{
			if (!FPSRigidBodyWalker.code.isFloating)
			{
				FPSRigidBodyWalker.code.isUnderWater = true;
				EnviroManager.instance.PlayerUnderWater = true;
				underwaterSnapshot.TransitionTo(0.2f);
				if (InputManager.Instance.InteractSecondaryPressed && (bool)rebreather && rebreather.Condition > 0f && !breathing)
				{
					breathing = true;
				}
				if (breathing)
				{
					rebreatherTimer += Time.deltaTime;
					if (rebreatherTimer >= 0f && rebreatherTimer < 0.4f)
					{
						rebreather.Condition -= Global.code.ConstantData.AirTankConsumingOnce / 0.4f * Time.deltaTime;
						if (rebreather.Condition <= 0f)
						{
							rebreather.Condition = 0f;
						}
						if (BreatheState != 1)
						{
							BreatheState = 1;
							FPSPlayer.code.ToggleBreathAudio(state: true);
						}
					}
					else if (rebreatherTimer > 0.4f && rebreatherTimer < 1f)
					{
						float num = Mathf.Floor(rebreather.Condition / Global.code.ConstantData.AirTankConsumingOnce);
						rebreather.Condition = num * Global.code.ConstantData.AirTankConsumingOnce;
						if (rebreather.Condition <= 0f)
						{
							rebreather.Condition = 0f;
						}
						Air += Global.code.ConstantData.AirTankConsumingOnce / 0.7f * Time.deltaTime * Global.code.ConstantData.AirTankRatio;
						if (BreatheState != 2)
						{
							FPSPlayer.code.ToggleBreathAudio(state: false);
							BreatheState = 2;
						}
					}
					else
					{
						BreatheState = 0;
						rebreatherTimer = 0f;
						FPSPlayer.code.ShowBreathBubble(showBubble: false);
						breathing = false;
					}
					if (Air >= MaxAir)
					{
						Air = MaxAir;
					}
				}
				else
				{
					rebreatherTimer = 0f;
					BreatheState = 0;
					FPSPlayer.code.ShowBreathBubble(showBubble: false);
				}
				float num2 = 1f;
				if ((bool)rebreather && rebreather.ItemID == RM.code.RecyclableAirTank.ItemID)
				{
					num2 = 0.25f;
				}
				Air -= Time.deltaTime * AirConsumtionRate * depthAirConsumptionRatio * num2;
				if (Air <= MaxAir * 0.1f && (bool)rebreather && rebreather.Condition > 0f && !breathing && autoBreath)
				{
					breathing = true;
				}
				audioSourceUnderwater.volume = Mathf.Lerp(audioSourceUnderwater.volume, 1f, Time.deltaTime);
				DistinguishTorch();
			}
		}
		else
		{
			if (FPSRigidBodyWalker.code.isUnderWater && !RM.code.aud.isPlaying && drivingObject == null && seatObject == null)
			{
				RM.code.PlayOneShot(sndExitWater, UnityEngine.Random.Range(0.9f, 1.1f));
			}
			FPSRigidBodyWalker.code.isUnderWater = false;
			EnviroManager.instance.PlayerUnderWater = false;
			onLandSnapshot.TransitionTo(0.2f);
			Air += Time.deltaTime * AirConsumtionRate * 10.4f;
			Air = Mathf.Clamp(Air, 0f, MaxAir);
			audioSourceUnderwater.volume = Mathf.Lerp(audioSourceUnderwater.volume, 0f, Time.deltaTime);
		}
		if (depth < 0.4f || (bool)CurAquarium)
		{
			MainCamera.nearClipPlane = 0.08f;
			if (!FPSRigidBodyWalker.code.isSwimming)
			{
				FPSRigidBodyWalker.code.isSwimming = true;
			}
			Wet = 30f;
		}
		else
		{
			RPGCamera.code.cam.nearClipPlane = 0.02f;
			if (FPSRigidBodyWalker.code.isSwimming)
			{
				FPSRigidBodyWalker.code.isSwimming = false;
			}
			Wet = Mathf.Lerp(Wet, 0f, Time.deltaTime * 0.1f);
		}
		if (HasNoAir())
		{
			Air = 0f;
			AddHealth((0f - Time.deltaTime) * 3.5f, 3, Vector3.zero, canBleed: false, showBiteFx: false, IsMeleeWeapon: false, ignoreArmor: false, isGrenade: false, PlayerDeathType.OutOfOxygen);
			if (UnityEngine.Random.Range(0, 100) < 1 && !RM.code.aud.isPlaying)
			{
				RM.code.PlayOneShot(sndDrowning[IsFemale ? 1 : 0], UnityEngine.Random.Range(0.9f, 1.1f));
			}
			drowningEffect.weight = Mathf.Lerp(drowningEffect.weight, 1f, Time.deltaTime * 0.15f);
		}
		else
		{
			drowningEffect.weight = Mathf.Lerp(drowningEffect.weight, 0f, Time.deltaTime * 0.7f);
		}
		if (!weaponInHand || !weaponInHand.TryGetComponent<WaterContainer>(out var component))
		{
			return;
		}
		if (FPSRigidBodyWalker.code.isInWater)
		{
			if (component._item.Condition < component._item.MaxCondition && !FocusedInteraction)
			{
				Global.code.uiCombat.ShowHintWithKeycodePenal(InputManager.Instance.InteractKeyAction.keyCode, MyBuildController.FillSeaWater.GetLocalizedString());
				if (InputManager.Instance.InteractPressed)
				{
					component._item.Condition = component._item.MaxCondition;
					RM.code.PlayOneShot(sndFillSeawaterBucket, UnityEngine.Random.Range(0.9f, 1.1f));
					Global.code.uiCombat.RefreshQuickSlot();
					Global.code.uiCombat.HideHint();
					PlayerWeapons.code.CurrentWeaponBehaviorComponent.ReFreshWaterBucketState();
				}
			}
			else
			{
				Global.code.uiCombat.HideHintWithKeycodePenal();
			}
		}
		else
		{
			Global.code.uiCombat.HideHintWithKeycodePenal();
		}
	}
